def write(path, value):
    with path.open('x', encoding='utf-8') as stream:
        json.dump(value, stream, ensure_ascii=False, indent=2, allow_nan=False)
        stream.write('\n')


def execute(name, args, stage):
    if subprocess.check_output(['git','status','--porcelain'], text=True).strip():
        raise RuntimeError('Worktree must remain clean for scientific execution')
    command = [sys.executable, *args]
    print('Running:', ' '.join(command), flush=True)
    output = (base/f'{name}.log').read_text()
    # These commands already completed in attempt 1; verified_run below rechecks their chains.
    receipt['commands'].append({'argv': command, 'exit_code': 0,
                                'execution_reused_from_completed_command': True,
                                'log': str((base/f'{name}.log').relative_to(root)),
                                'sha256': sha256_file(base/f'{name}.log')})
    match = re.search(r'SUCCESS: (.+)', output)
    if not match:
        raise RuntimeError(f'Run identifier not found in {name}')
    run_id = Path(match.group(1).strip()).name
    manifest, files, execution = verified_run(root, run_id, stage)
    assert manifest['dirty_worktree'] is False and execution['scientific_eligible'] is True
    print('Completed:', run_id, flush=True)
    return run_id, files


def compare(old, new, exclude=()):
    left, right = read_table(old), read_table(new)
    assert len(left) == len(right), (old, 'row count')
    changes = []
    for number, (a, b) in enumerate(zip(left, right, strict=True), 1):
        assert set(a) == set(b), (old, 'schema')
        for field in a:
            if field not in exclude and a[field] != b[field]:
                changes.append({'row': number, 'commit_sha': a.get('commit_sha'),
                                'file_path': a.get('file_path'), 'field': field,
                                'before': a[field], 'after': b[field]})
    receipt['comparisons'].append({'old': str(old.relative_to(root)), 'new': str(new.relative_to(root)),
                                   'old_sha256': sha256_file(old), 'new_sha256': sha256_file(new),
                                   'rows': len(left), 'excluded_metadata_fields': list(exclude),
                                   'differences': changes})
    return changes
